    <div class="card-row">

        <el-card v-for="item in spots" :key="item.name" class="poem-card" @click="showDetail(item)">

            <img :src="item.img" style="width:100%">

            <h4>{{item.name}} · {{item.poet}}</h4>

            <p style="font-size: 12px; color: #666;">{{item.poem}}</p>

        </el-card>

    </div>

 

    <el-dialog v-model="dialogVisible" :title="activeSpot.poem" width="80%" center>

        <div style="text-align: center;">

            <el-tag type="danger">{{activeSpot.tag}}</el-tag>

            <p><strong>白话翻译：</strong>{{activeSpot.desc}}</p>

        </div>

        <template #footer>

            <el-button type="primary" @click="dialogVisible = false">知道了</el-button>

        </template>

    </el-dialog>

 

    <el-menu mode="horizontal" class="nav-footer" default-active="1" border-top>

        <el-menu-item index="1">首页</el-menu-item>

    </el-menu>

<script>

    const { createApp, ref, onMounted } = Vue;

 

    createApp({

        setup() {

            const dialogVisible = ref(false);

            const activeSpot = ref({});

            const spots = ref([

                { name: '黄鹤楼', poet: '李白', poem: '《黄鹤楼送孟浩然之广陵》', tag: '豪迈', desc: '李白站在黄鹤楼上，目送友人乘船东下...', img: 'https://via.placeholder.com/150' },

                { name: '西湖', poet: '白居易', poem: '《钱塘湖春行》', tag: '清新', desc: '早春的西湖，生机勃勃...', img: 'https://via.placeholder.com/150' }

            ]);

 

            const showDetail = (item) => {

                activeSpot.value = item;

                dialogVisible.value = true;

            };

 

            onMounted(() => {

                // 初始化地图 (界面1 )

                const chart = echarts.init(document.getElementById('map-container'));

                // 注意：正式比赛需加载真实的 china.json

                fetch('https://geo.datav.aliyun.com/areas_v3/bound/100000_full.json')

                    .then(res => res.json())

                    .then(geoJson => {

                        echarts.registerMap('china', geoJson);

                        chart.setOption({

                            geo: { map: 'china', itemStyle: { areaColor: '#c1d8ac', borderColor: '#2f4f4f' } },

                            series: [{

                                type: 'scatter',

                                coordinateSystem: 'geo',

                                data: [{ name: '武汉', value: [114.31, 30.52] }],

                                itemStyle: { color: 'red' }

                            }]

                        });

                    });

            });

 

            return { spots, dialogVisible, activeSpot, showDetail };

        }

    }).use(ElementPlus).mount('#app');

</script>
